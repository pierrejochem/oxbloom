use oxbloom_boot::OxbloomApplication;

#[tokio::main]
async fn main() -> ExitCode {
    OxbloomApplication::builder()
        .args(std::env::args().skip(1))
        .config(|b| b.dir("config"))
        .run_and_exit()
        .await
}
